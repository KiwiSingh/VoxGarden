cmake_minimum_required(VERSION 3.24)
project(VoxGarden VERSION 1.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "")
set(VOX_DEPS "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty" CACHE PATH "Dependency checkout directory")
add_subdirectory(${VOX_DEPS}/JUCE JUCE)
add_subdirectory(${VOX_DEPS}/linear linear)
add_subdirectory(${VOX_DEPS}/signalsmith-stretch stretch)
juce_add_plugin(VoxGarden COMPANY_NAME "Kiwi Singh" BUNDLE_ID "audio.voxgarden.choir" PLUGIN_MANUFACTURER_CODE Vxga PLUGIN_CODE Vxg1 FORMATS VST3 AU Standalone PRODUCT_NAME "VoxGarden" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx PitchShift AU_MAIN_TYPE kAudioUnitType_MusicEffect)
juce_add_binary_data(VoxGardenSingingResources HEADER_NAME SingingResources.h NAMESPACE SingingData SOURCES Singing/runtime.py Singing/bootstrap.sh Singing/model-manifest.json)
target_link_libraries(VoxGarden PUBLIC VoxGardenSingingResources juce::juce_cryptography)
target_sources(VoxGarden PRIVATE Source/Processor.cpp Source/Editor.cpp)
target_include_directories(VoxGarden PUBLIC Source)
target_compile_definitions(VoxGarden PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VoxGarden PUBLIC juce::juce_audio_utils juce::juce_dsp signalsmith-stretch PRIVATE juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(VoxGardenCheck PRODUCT_NAME "VoxGardenCheck")
target_sources(VoxGardenCheck PRIVATE tests/Checks.cpp)
target_compile_definitions(VoxGardenCheck PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=1)
target_link_libraries(VoxGardenCheck PRIVATE VoxGarden)
enable_testing()
add_test(NAME DSP_and_state COMMAND VoxGardenCheck)
# This effect uses note input, not thousands of synthetic MIDI-CC parameters.
# Keep SnappySnap and Live's Configure list focused on the 54 sound controls.
target_compile_definitions(VoxGarden_VST3 PRIVATE JUCE_VST3_EMULATE_MIDI_CC_WITH_PARAMETERS=0)
option(VOX_BUILD_SNAP_TEST_HOST "Build isolated SnappySnap GUI compatibility harness" OFF)
if(VOX_BUILD_SNAP_TEST_HOST)
  juce_add_gui_app(VoxGardenSnapTest PRODUCT_NAME "VoxGarden Snap Test" BUNDLE_ID "audio.voxgarden.snaptest")
  target_sources(VoxGardenSnapTest PRIVATE tests/SnappyHost.cpp)
  target_compile_definitions(VoxGardenSnapTest PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  target_link_libraries(VoxGardenSnapTest PRIVATE juce::juce_audio_utils)
endif()
if(VOX_BUILD_SNAP_TEST_HOST)
  set_target_properties(VoxGardenSnapTest PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
